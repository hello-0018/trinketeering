*TEMPLATE
text:[this]

*snowglobe
name:Snowglobe|Snowglobes
icon:https://i.ibb.co/tTtkyNS9/trinket-1-png.png
desc:Click for details<q>A cozy scene quite literally frozen in time.</q>
cost:100 coolness, 5 snowflake, 1 woodPiece, 3 glassShard
on click:
    clear log
    log <b>Snowglobe</b>
    log Per Second:
    log <.>Yields 1 Coolness
    log On Shake:
    log <.>[?(snowglobeShake)|Yields 10 Coolness|<i>Requires [n:snowglobeShake]</i>]
    log <.>[?(snowglobeShake)|(1%) Yields 1 Snowflake|<i>Requires [n:snowglobeShake]</i>]
    log <.>[?(snowfall)|Yields 1 Coolness|<i>Requires [n:snowfall]</i>]
    log <.>[?(snowfall2)|Yields 2 Coolness|<i>Requires [n:snowfall2]</i>]
    log <.>[?(snowfall3)|Yields 7 Coolness|<i>Requires [n:snowfall3]</i>]
end
on tick:
    yield 1 coolness
end
on shake:
    if (snowglobeShake)
        yield 10 coolness
        if (chance(1)) yield 1 snowflake
    end
end
on snowflakeClicked:
    if (snowfall) yield 1 coolness
    if (snowfall2) yield 2 coolness
    if (snowfall3) yield 7 coolness
end

*spinningTop
name:Spinning Top|Spinning Tops
icon:https://i.ibb.co/1YnwzQV5/trinket-9-png.png
desc:Click for details<q>Its very movement keeps it alive.</q>
cost:100 zest, 5 woodPiece
req:spinPark
on click:
    clear log
    log <b>Spinning Top</b>
    log Passive:
    log <.>Grants 5 Rotativity
    log On Rotate:
    log <.>Yields 30 Zest
    log <.>[?(momentum)|(15%) Increases Rotativity by 1|<i>Requires [n:momentum]</i>]
end
passive:
    grant 5 rotativity
end
on rotate:
    yield 30 zest
    if (momentum and chance(15))yield 1 rotativity
end

Upgrades

*system
hidden
start with
tags:Prestige
on tick:
    if (chance(pow(shakiness, 1/4))) spawn shakePrompt
    if (chance(pow(rotativity, 1/4))) spawn rotatePrompt
end
passive:
    if (!(have snowglobe)) multiply cost of snowglobe by 0
end
